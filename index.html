VBA — Auto Weekly Report Generator 
Sub GenerateWeeklyReport()
    ' Auto-generate weekly progress report from site data
    Dim ws As Worksheet
    Dim lastRow As Long
    Dim totalProgress As Double

    Set ws = ThisWorkbook.Sheets("SiteData")
    lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row

    ' Loop through all activities and calculate weighted progress
    For i = 2 To lastRow
        totalProgress = totalProgress + _
            ws.Cells(i, "D").Value * ws.Cells(i, "E").Value
    Next i

    ' Write to report sheet and format
    Sheets("WeeklyReport").Range("B5").Value = _
        Format(totalProgress / 100, "0.00%")
    MsgBox "Report generated successfully!"
End Sub
